def ident(b):
    return {"bytes":len(b),"sha256":hashlib.sha256(b).hexdigest(),"git_blob":hashlib.sha1(b"blob "+str(len(b)).encode()+b"\0"+b).hexdigest()}
def index(tree):
    result={}
    for n in tree.body:
        key=getattr(n,"name",None)
        if isinstance(n,ast.Assign) and isinstance(n.targets[0],ast.Name):key=n.targets[0].id
        if isinstance(n,ast.AnnAssign) and isinstance(n.target,ast.Name):key=n.target.id
        if key: result[key]=n
    return result
