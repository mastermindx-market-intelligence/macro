ROOT = Path(__file__).resolve().parents[2]

def source(path):
    return subprocess.run(["git", "show", f"{PIN}:{path}"], cwd=ROOT,
                          check=True, capture_output=True, timeout=30).stdout.decode()

def producer():
    text = source("scripts/build_china.py")
    fn = next(n for n in ast.parse(text).body if isinstance(n, ast.FunctionDef) and n.name == "_radar_dlg_vm")
    maps = [n for n in fn.body if isinstance(n, ast.Assign) and any(
        isinstance(t, ast.Name) and t.id in {"_REC_READ", "_DD_READ"} for t in n.targets)]
    blocks = [n for n in fn.body if isinstance(n, ast.Try) and any(
        isinstance(k, ast.Name) and k.id == "_DD_READ" for k in ast.walk(n))]
    if len(maps) != 2 or len(blocks) != 1:
        raise RuntimeError("Gauge producer block is not uniquely identified")
    return compile(ast.Module(body=[*maps, *blocks], type_ignores=[]), "actual_gauge_producer", "exec"), text
