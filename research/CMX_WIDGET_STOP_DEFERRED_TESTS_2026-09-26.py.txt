from __future__ import annotations

ROOT = Path(__file__).resolve().parents[1]
SOURCE = ROOT / "templates" / "mm_brain.js"


def _function(name: str) -> str:
    text = SOURCE.read_text(encoding="utf-8")
    start = text.index("  function " + name + "(")
    # Top-level IIFE function boundaries use two spaces; nested blocks use >=4.
    end = text.index("\n  }", start) + len("\n  }")
    return text[start:end]


def _node(names: list[str], program: str) -> None:
    node = shutil.which("node")
    if not node:
        pytest.skip("node required for real JavaScript function evaluation")
    code = "const assert = require('node:assert/strict');\n" + "\n".join(_function(n) for n in names)
    code += "\n(async function () {\n" + program + "\n})().catch(e => { console.error(e); process.exitCode=1; });"
    result = subprocess.run([node, "-e", code], cwd=ROOT, text=True, capture_output=True, timeout=10)
    assert result.returncode == 0, result.stdout + result.stderr


@pytest.mark.parametrize("flag", ["stopped", "retracted", "doneSeen"])
def test_no_chart_handoff_from_inactive_reply(flag: str) -> None:
    _node(["newTurn", "forwardChartEvent"], r"""
      global.CFG = {onCommand: () => { throw Error('must not execute'); }};
      global.saveRun = () => {}; global.setBusy = () => {};
      const t = newTurn({}, null); t[FLAG] = true;
      forwardChartEvent({type:'command',v:2,batch_id:'brain_a'}, t);
      assert.equal(t.chartHandoff, false);
    """.replace("FLAG", json.dumps(flag)))


def test_handoff_is_recorded_before_a_throwing_callback() -> None:
    _node(["newTurn", "forwardChartEvent"], r"""
      const order = [];
      global.saveRun = t => { order.push('persist'); assert.equal(t.chartHandoff, true); };
      global.setBusy = () => {};
      global.CFG = {onCommand: () => {order.push('callback'); throw Error('possibly applied');}};
      const t = newTurn({}, null);
      forwardChartEvent({type:'command',v:2,batch_id:'brain_a'}, t);
      assert.deepEqual(order, ['persist','callback']);
      assert.equal(t.chartHandoff, true);
      assert.deepEqual(t.chartBatches, ['brain_a']);
      assert.equal(t.chartHandoffUnconfirmed, true);
    """)

    _node(["newTurn", "stopChartWork"], r"""

def test_buffered_late_chunk_does_not_reach_host_after_stop() -> None:

      const t=newTurn({},null);global.activeStream=t;
      let forwarded=0,cancelled=0,read=0;
      global.forwardChartEvent=()=>{forwarded++;t.stopped=true;};
      global.saveRun=()=>{};global.finish=()=>{};
      const bytes=new TextEncoder().encode('data: {"type":"command"}\n\ndata: {"type":"command"}\n\n');
      const reader={read:()=>Promise.resolve(read++ ? {done:true}:{done:false,value:bytes}),cancel:()=>{cancelled++;return Promise.resolve();}};
      await readSse({body:{getReader:()=>reader}},t);
      assert.equal(forwarded,1);assert.ok(cancelled>=1);
    """)
