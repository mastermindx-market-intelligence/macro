"""Synthetic native TrialLedger and proposed-manifest contract checks.
No real TOI configs, production ledger, market observations, or outcomes are used.
Run with SUBJECT_ROOT pointing at an immutable source export and INVENTORY_PATH
pointing at the existing 147-slot draft. Native source is never edited.
"""
import json, os, sys, tempfile, unittest
from pathlib import Path
from collections import Counter
sys.dont_write_bytecode = True
sys.path.insert(0, os.environ['SUBJECT_ROOT'])
from engine import trial_ledger as native

IDS = [1,2,5,6,9,10,12,13,15,16,17,19,20,23,24,27,28,30,31,33,34,35]
FAMILY = '__toi_offline_contract_fixture__'

class LedgerContract(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory(prefix='toi-ledger-fixture-')
        self.root = Path(self.tmp.name)
        self.path = self.root/'synthetic.jsonl'
        self.old_default = native.DEFAULT_PATH
        native.DEFAULT_PATH = self.root/'FORBIDDEN_DEFAULT.jsonl'
        self.led = native.TrialLedger(path=self.path, family=FAMILY)
        self.grid = [{'synthetic_index':i,'fixture_only':True} for i in range(22)]
    def tearDown(self):
        self.assertFalse(native.DEFAULT_PATH.exists(), 'default ledger must never be opened')
        native.DEFAULT_PATH = self.old_default
        self.tmp.cleanup()
    def test_22_itemized_configs_count_22(self):
        self.assertEqual(self.led.log_grid(self.grid),22)
        self.assertEqual((self.led.literal_n(),self.led.effective_n()),(22,22))
    def test_budget_is_floor_not_addition(self):
        self.led.log_declared_budget(22, reason='synthetic whole-grid floor')
        self.led.log_grid(self.grid)
        self.assertEqual(self.led.effective_n(),22)
        self.assertNotEqual(self.led.effective_n(),44)
    def test_reverse_budget_order_same_result(self):
        self.led.log_grid(self.grid)
        self.led.log_declared_budget(22, reason='synthetic whole-grid floor')
        self.assertEqual(self.led.effective_n(),22)
    def test_rerun_and_restart_idempotent(self):
        self.led.log_grid(self.grid)
        before=self.path.read_bytes()
        restored=native.TrialLedger(path=self.path,family=FAMILY)
        self.assertEqual(restored.log_grid(self.grid),0)
        self.assertEqual(restored.effective_n(),22)
        self.assertEqual(self.path.read_bytes(),before)
    def test_dictionary_key_order_does_not_create_new_trial(self):
        self.assertTrue(self.led.log_trial({'x':1,'z':{'b':2,'a':3}}))
        self.assertFalse(self.led.log_trial({'z':{'a':3,'b':2},'x':1}))
    def test_metadata_only_change_is_not_hashed_configuration(self):
        cfg={'fixture_only':True,'model':'same'}
        self.assertTrue(self.led.log_trial(cfg,source='fixture-source-A',info_cutoff='2020-01-01'))
        self.assertFalse(self.led.log_trial(cfg,source='fixture-source-B',info_cutoff='2020-02-01'))
        self.assertEqual(self.led.literal_n(),1)
    def test_config_bound_data_version_is_a_distinct_trial(self):
        for vintage in ['fixture-A','fixture-B']:
            self.assertTrue(self.led.log_trial({'fixture_only':True,'data_version':vintage}))
        self.assertEqual(self.led.literal_n(),2)
    def test_counted_policy_endpoint_variants_do_not_collapse(self):
        for terminal in ['fixture-common-terminal','fixture-fill-relative']:
            self.led.log_trial({'fixture_only':True,'policy':{'terminal':terminal}})
        self.assertEqual(self.led.literal_n(),2)
    def test_historical_family_stays_separate(self):
        self.led.log_grid(self.grid)
        self.led.log_declared_budget(12,family='__s16_history_fixture__',reason='synthetic only')
        self.assertEqual(self.led.effective_n(FAMILY),22)
        self.assertEqual(self.led.effective_n('__s16_history_fixture__'),12)
    def test_context_budget_precedes_body_with_explicit_ledger(self):
        with native.register_trials(FAMILY,budget=22,ledger=self.led,reason='synthetic'):
            self.assertEqual(self.led.declared_budget(),22)
            self.assertEqual(self.led.literal_n(),0)
    def test_decorator_budget_precedes_body_with_explicit_ledger(self):
        @native.register_trials(FAMILY,budget=22,ledger=self.led,reason='synthetic')
        def body():
            return self.led.declared_budget()
        self.assertEqual(body(),22)
    def test_lower_budget_and_correlation_credit_cannot_erase_floor(self):
        self.led.log_grid(self.grid)
        self.led.log_declared_budget(22)
        self.led.log_declared_budget(1)
        self.assertEqual(self.led.effective_n(correlation_credit=.01),22)
    def test_empty_effective_n_is_not_registration_evidence(self):
        self.assertEqual((self.led.literal_n(),self.led.declared_budget()),(0,0))
        self.assertEqual(self.led.effective_n(),1)
        self.assertFalse(self.path.exists())
    def test_invalid_zero_budget_refuses_before_body(self):
        with self.assertRaises(ValueError):
            with native.register_trials(FAMILY,budget=0,ledger=self.led):
                self.fail('invalid budget entered body')

class ProposedInventoryContract(unittest.TestCase):
    def setUp(self):
        self.inv=json.loads(Path(os.environ['INVENTORY_PATH']).read_text())
        self.by_id={r['planning_slot']:r for r in self.inv['objects']}
        self.subset=[self.by_id[f'PLAN-{n:03d}'] for n in IDS]
    def test_original_147_slots_are_untouched_and_unadmitted(self):
        self.assertEqual(len(self.by_id),147)
        for obj in self.by_id.values():
            self.assertFalse(obj['selected_for_execution'])
            self.assertFalse(obj['registered'])
            self.assertEqual(obj['status'],'DRAFT_NOT_ADMITTED')
            self.assertIsNone(obj['native_scientific_or_trial_id'])
    def test_22_proposed_slots_are_balanced_daily_only(self):
        self.assertEqual(len(set(IDS)),22)
        self.assertEqual(Counter(r['direction'] for r in self.subset),{'UP':11,'DOWN':11})
        self.assertEqual({r['clock_scope'] for r in self.subset},{'DAILY'})
    def test_exact_object_mix_and_eight_contrasts(self):
        self.assertEqual(Counter(r['configuration_kind'] for r in self.subset),{
          'core_construction':4,'simple_breakout_comparator':2,'near_boundary_placebo':2,
          'confirmation_policy_application':4,'activation_model_specification':4,
          'post_trigger_model_specification':6})
        claims=self.inv['proposed_primary_contrasts']
        self.assertEqual(len(claims),8)
        self.assertEqual(Counter(c['direction'] for c in claims),{'UP':4,'DOWN':4})
        self.assertTrue(all(c['native_claim_id'] is None for c in claims))

if __name__ == '__main__':
    unittest.main(verbosity=2)
